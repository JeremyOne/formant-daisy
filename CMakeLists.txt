cmake_minimum_required(VERSION 3.22)
# Version: bump the minor number for every change; tag vX.Y.Z to publish a release.
project(FormantDaisy VERSION 0.15.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "")

# JUCE: use a local checkout if JUCE_DIR is set, otherwise download it.
set(JUCE_DIR "" CACHE PATH "Path to a JUCE checkout (optional)")
if(JUCE_DIR AND EXISTS "${JUCE_DIR}/CMakeLists.txt")
    add_subdirectory(${JUCE_DIR} JUCE)
else()
    include(FetchContent)
    FetchContent_Declare(JUCE
        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
        GIT_TAG 8.0.4
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()

# The VST3 moduleinfo.json manifest is generated by running a helper on the
# build machine. Turn it off when cross-compiling (e.g. Windows from Linux).
option(FD_VST3_MANIFEST "Generate VST3 moduleinfo.json" ON)

# Formats: VST3 everywhere, AU on macOS, plus a standalone app for quick testing.
set(FD_FORMATS VST3 Standalone)
if(APPLE)
    list(APPEND FD_FORMATS AU)
endif()

juce_add_plugin(FormantDaisy
    COMPANY_NAME "Project Mammoth"
    PRODUCT_NAME "Formant Daisy"
    BUNDLE_ID "com.projectmammoth.formantdaisy"
    PLUGIN_MANUFACTURER_CODE Hmbw
    PLUGIN_CODE FDsy
    FORMATS ${FD_FORMATS}
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE
    COPY_PLUGIN_AFTER_BUILD FALSE
    VST3_CATEGORIES Instrument Synth
    VST3_AUTO_MANIFEST ${FD_VST3_MANIFEST}
    AU_MAIN_TYPE kAudioUnitType_MusicDevice)

target_sources(FormantDaisy PRIVATE
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
    Source/PixelFont.cpp
    Source/engine/Phonemes.cpp
    Source/engine/TextToPhonemes.cpp
    Source/engine/FormantSinger.cpp
    Source/engine/CmuDict.cpp)

target_include_directories(FormantDaisy PRIVATE Source)

target_compile_definitions(FormantDaisy PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0)

target_link_libraries(FormantDaisy
    PRIVATE
        juce::juce_audio_utils
        juce::juce_dsp
    PUBLIC
        juce::juce_recommended_config_flags
        juce::juce_recommended_lto_flags
        juce::juce_recommended_warning_flags)

# Offline test tools (no JUCE needed)
add_executable(render_daisy tools/render_daisy.cpp
    Source/engine/Phonemes.cpp Source/engine/TextToPhonemes.cpp Source/engine/FormantSinger.cpp Source/engine/CmuDict.cpp)
target_include_directories(render_daisy PRIVATE Source)

# Dev-only: headless UI screenshot tool (not part of the plugin)
option(FD_BUILD_DEV_TOOLS "Build developer tools" OFF)
if(FD_BUILD_DEV_TOOLS)
    juce_add_console_app(ui_snapshot PRODUCT_NAME "ui_snapshot")
    target_sources(ui_snapshot PRIVATE tools/ui_snapshot.cpp
        Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/PixelFont.cpp
        Source/engine/Phonemes.cpp Source/engine/TextToPhonemes.cpp Source/engine/FormantSinger.cpp Source/engine/CmuDict.cpp)
    target_include_directories(ui_snapshot PRIVATE Source)
    target_compile_definitions(ui_snapshot PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
        JucePlugin_Name="Formant Daisy" JucePlugin_VersionString="${PROJECT_VERSION}" JUCE_MODAL_LOOPS_PERMITTED=1)
    target_link_libraries(ui_snapshot PRIVATE juce::juce_audio_utils juce::juce_dsp)
endif()
